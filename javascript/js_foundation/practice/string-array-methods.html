Basic String & Array Methods

Useful built-in methods to work with strings and arrays in JavaScript

1. String Methods

let text = "  Hello JavaScript!  ";

console.log(text.length);         // 21
console.log(text.toUpperCase()); // "  HELLO JAVASCRIPT!  "
console.log(text.toLowerCase()); // "  hello javascript!  "
console.log(text.trim());        // "Hello JavaScript!"
console.log(text.includes("Java")); // true
console.log(text.indexOf("J"));  // 8
console.log(text.slice(2, 7));   // "Hello"

2. Array Methods

let nums = [1, 2, 3, 4, 5];

console.log(nums.join("-"));       // "1-2-3-4-5"
console.log(nums.includes(3));     // true
console.log(nums.indexOf(4));      // 3
console.log(nums.concat([6, 7]));  // [1, 2, 3, 4, 5, 6, 7]
console.log(nums.reverse());       // [5, 4, 3, 2, 1]
console.log(nums.slice(1, 4));     // [4, 3, 2]
console.log(nums.splice(2, 1));    // Removes 1 item at index 2

3. Split & Join (String ↔ Array)

let sentence = "JavaScript is awesome!";
let words = sentence.split(" ");
console.log(words); // ["JavaScript", "is", "awesome!"]

let joined = words.join(" ");
console.log(joined); // "JavaScript is awesome!"
